### Скрипт пункта 3

```python
import re
import pathlib

text = pathlib.Path("v4/03b-mlp-transformer.md").read_text(encoding="utf-8")
text = text.replace("\r\n", "\n")
speech = text.split("\n---\n")[1]
paras = [p for p in speech.split("\n\n") if p.strip()]
violations = []
for i, para in enumerate(paras, 1):
    cleaned = re.sub(r"\[(?:КАРТА|ЗЛОБА):[^\]]*\]", " ", para)
    cleaned = cleaned.replace("…", "\uE000")
    parts = [
        s.replace("\uE000", "…").strip()
        for s in re.split(r"(?<=[.!?])\s+", cleaned.strip())
        if s.strip()
    ]
    run = []
    for sent in parts:
        n = len(re.findall(r"[0-9A-Za-zА-Яа-яЁё]+", sent))
        if n < 5:
            run.append((n, sent))
            if len(run) >= 3:
                violations.append((i, list(run)))
        else:
            run = []
print("paragraphs", len(paras))
print("violations", len(violations))
for item in violations:
    print(item)
```
